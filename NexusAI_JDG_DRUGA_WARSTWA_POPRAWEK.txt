  import duckdb

  def execute_with_memory_bound(query: str, tenant_id: str):
      """Zapytanie DuckDB z limitem pamięci per tenant (512 MB).
      Nie używamy mem_alloc (nierealistyczne); używamy mechanizmu
      DuckDB memory_limit ustawianego per connection."""
      conn = duckdb.connect(f":memory:")
      # DuckDB posiada natywny statement memory_limit (od 0.8.0)
      conn.execute("SET memory_limit='512MB'")
      conn.execute("SET threads=4")        # Limit wątków
      result = conn.execute(query).fetchall()
      conn.close()                        # Wymusza zwolnienie pamięci
      return result
  ```

  # Bezpieczny merge chroniący ZUS P720 przed nadpisaniem przez risk.decide.
  # Helper definicja (potrzebna do działania safe_merge):
  has_immutable_flag(v) {
      v.immutable_verdict == true
  }

  safe_merge(a, b) = c {
      has_immutable_flag(a)
      c := a  # Immutable verdict wins, nie można nadpisać
      # nawet przez wyższy priorytet (risk, routing)

  safe_merge(a, b) = object.union(b, a) {
      not has_immutable_flag(a)
  }

  # Zastosoje w final_verdict (zamist object.union chain):
  final_verdict = safe_merge(risk.decide,
      safe_merge(routing.decide,
      safe_merge(compliance.decide,
      ...
          fallback.decide
      ))))...
  ```

  class ImmutableAuditWithRODO:
      def __init__(self):
          self.kms = KMSHashicorp.instance()
      def process_audit_record(self, payload):
          # NIE przechowujemy PESEL/NIP, tylko HASH
          salt = self.kms.generate_salt(payload.tenant_id)
          anonymized_hash = hashlib.sha256(
              (payload.pesel + salt).encode()
          ).hexdigest()
          merkle_tree.append(anonymized_hash)
      def forget_tenant(self, tenant_id):
          """Art. 17 RODO: Prawo do bycia zapomnianym."""

          # Merkle node zostaje, ale jest
